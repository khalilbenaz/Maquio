# Calque v1 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Livrer une application de bureau qui dessine des interfaces mobiles au drag & drop, importe des maquettes Figma, se fait assister par Claude Code, et exporte le résultat en Flutter, React Native, SwiftUI et Compose.

**Architecture:** Monorepo npm workspaces. Quatre paquets de logique pure en TypeScript (`core`, `figma`, `codegen`, `ai`) qui ne connaissent ni Electron ni React, et une application Electron (`apps/desktop`) qui les assemble. Toute mutation du document passe par des commandes annulables ; les patchs de Claude Code empruntent exactement ce chemin.

**Tech Stack:** TypeScript 5.6 strict, Node 24, Vitest 2, Zod 3, React 18, Vite 5, Electron 32, electron-builder, Testing Library + jsdom.

**Spec:** `docs/superpowers/specs/2026-09-20-calque-design.md`

## Global Constraints

- **Règle de dépendance** : `packages/core` n'importe aucun autre paquet du dépôt. `packages/figma`, `packages/codegen` et `packages/ai` n'importent que `@calque/core`. Seul `apps/desktop` importe `electron` ou `react`. Un test l'impose (Tâche 18).
- **Hermétisme des tests** : `npm test` s'exécute sans écran, sans réseau, sans SDK Flutter/Dart, sans binaire `claude`. Tout accès au monde extérieur (fetch, spawn, fs) est injecté par interface et remplacé par un double dans les tests.
- **TypeScript strict** : `"strict": true`, `"noUncheckedIndexedAccess": true`, aucun `any` implicite, aucun `@ts-ignore`.
- **Zod pinné en `^3.23.0`** : la v4 change la surface d'API (`.strict()`, messages d'erreur) ; la v1 ne s'y expose pas.
- **Electron** : `nodeIntegration: false`, `contextIsolation: true`, `sandbox: true`. Le renderer ne touche jamais au disque, au réseau ni à un sous-processus.
- **Version du format de document** : `DOCUMENT_VERSION = 1`. Un document d'une version supérieure est refusé, jamais lu partiellement.
- **Langue** : identifiants et code en anglais, commentaires et messages d'interface en français.
- **Commits** : un commit par tâche, message `type: sujet` en français sans accent dans le sujet.

// packages/core/src/tree/tree.test.ts
import { describe, expect, it } from 'vitest'
import { findNode, findParent, insertNode, removeNode, moveNode, absoluteFrame, hitTest } from './tree'
import type { FrameNode, Node } from '../model/types'

const leaf = (id: string, x: number, y: number): Node => ({
  id, name: id, type: 'rect', frame: { x, y, w: 50, h: 50 },
  visible: true, locked: false, opacity: 1, rotation: 0,
  fills: [{ type: 'solid', color: { r: 0, g: 0, b: 0, a: 1 } }], strokes: [], cornerRadius: 0,
})

const frame = (id: string, x: number, y: number, children: Node[]): FrameNode => ({
  id, name: id, type: 'frame', frame: { x, y, w: 200, h: 200 },
  visible: true, locked: false, opacity: 1, rotation: 0,
  layout: { mode: 'absolute', gap: 0, padding: { top: 0, right: 0, bottom: 0, left: 0 }, alignMain: 'start', alignCross: 'start' },
  fills: [], strokes: [], cornerRadius: 0, clipsContent: false, children,
})

const tree: Node[] = [frame('root', 10, 10, [leaf('a', 5, 5), frame('inner', 100, 100, [leaf('b', 1, 1)])])]

describe('parcours', () => {
  it('trouve un noeud profond et son parent', () => {
    expect(findNode(tree, 'b')?.id).toBe('b')
    expect(findParent(tree, 'b')?.id).toBe('inner')
    expect(findParent(tree, 'root')).toBeNull()
  })
  it('calcule un cadre absolu en cumulant les ancetres', () => {
    expect(absoluteFrame(tree, 'b')).toEqual({ x: 111, y: 111, w: 50, h: 50 })
  })
})

describe('mutations immuables', () => {
  it('insere sans muter l arbre d origine', () => {
    const next = insertNode(tree, 'inner', leaf('c', 0, 0))
    expect(findNode(next, 'c')).not.toBeNull()
    expect(findNode(tree, 'c')).toBeNull()
  })
  it('supprime un noeud profond', () => {
    expect(findNode(removeNode(tree, 'b'), 'b')).toBeNull()
  })
  it('reparente un noeud a la racine', () => {
    const next = moveNode(tree, 'b', null, 0)
    expect(findParent(next, 'b')).toBeNull()
    expect(next[0]!.id).toBe('b')
  })
  it('refuse de deplacer un noeud dans son propre descendant', () => {
    expect(() => moveNode(tree, 'root', 'inner', 0)).toThrow()
  })
})

describe('hitTest', () => {
  it('rend le noeud le plus profond sous le point', () => {
    expect(hitTest(tree, { x: 116, y: 116 })?.id).toBe('b')
  })
  it('ignore les noeuds verrouilles ou invisibles', () => {
    const verrouille: Node[] = [{ ...leaf('v', 0, 0), locked: true }]
    expect(hitTest(verrouille, { x: 10, y: 10 })).toBeNull()
  })
})

describe('History', () => {

// test/architecture.test.ts
import { readdirSync, readFileSync, statSync } from 'node:fs'
import { join } from 'node:path'
import { describe, expect, it } from 'vitest'

const fichiers = (dir: string): string[] =>
  readdirSync(dir).flatMap((e) => {
    const p = join(dir, e)
    if (e === 'node_modules' || e === 'dist') return []
    return statSync(p).isDirectory() ? fichiers(p) : p.match(/\.tsx?$/) ? [p] : []
  })

const imports = (f: string) =>
  [...readFileSync(f, 'utf8').matchAll(/from\s+['"]([^'"]+)['"]/g)].map((m) => m[1]!)

describe('regle de dependance', () => {
  it('core n importe aucun autre paquet du depot', () => {
    for (const f of fichiers('packages/core/src')) {
      for (const i of imports(f)) expect(i.startsWith('@calque/')).toBe(false)
    }
  })

  it('figma, codegen et ai n importent que core', () => {
    for (const p of ['figma', 'codegen', 'ai']) {
      for (const f of fichiers(join('packages', p))) {
        for (const i of imports(f)) {
          if (i.startsWith('@calque/')) expect(i).toMatch(/^@calque\/(core|codegen\/test)/)
        }
      }
    }
  })

  it('aucun paquet du coeur n importe electron ni react', () => {
    for (const p of ['core', 'figma', 'codegen', 'ai']) {
      for (const f of fichiers(join('packages', p))) {
        for (const i of imports(f)) expect(['electron', 'react', 'react-dom']).not.toContain(i)
      }
    }
  })

  it('le renderer n importe jamais node:fs ni node:child_process', () => {
    for (const f of fichiers('apps/desktop/src/renderer')) {
      for (const i of imports(f)) expect(i.startsWith('node:')).toBe(false)
    }
  })
})
